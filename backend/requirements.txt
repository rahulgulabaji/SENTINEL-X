# CPU/memory cost per guess. The one new auth-related dependency this
# project has; everything else here stays stdlib-only by design.
argon2-cffi==25.1.0

# HTTP client for the xAI Grok chatbot integration. sms_gateway.py's
# stdlib-urllib approach is fine for one fixed JSON POST, but the Grok
# chat-completions API needs multi-turn message arrays + structured JSON
# output -- httpx is the standard, well-maintained choice rather than
# hand-rolling that on urllib for a non-latency-critical chatbot path.
httpx==0.28.1
